// Compiled inside the pinned predecessor checkout by generate-admin-parent-fixture.sh.
package main

import (
	"context"
	"fmt"
	"os"

	"github.com/benhook1013/FireController/internal/backup"
	"github.com/benhook1013/FireController/internal/core"
)

func main() {
	if len(os.Args) != 3 {
		panic("expected source and bundle directories")
	}
	ctx := context.Background()
	store, err := core.Open(ctx, os.Args[1])
	if err != nil {
		panic(err)
	}
	defer store.Close()
	var live, domains int
	if err := store.DB().QueryRow(`SELECT count(*) FROM credentials WHERE revoked=0`).Scan(&live); err != nil || live == 0 {
		panic("parent must have a live credential")
	}
	if err := store.DB().QueryRow(`SELECT count(*) FROM metadata WHERE key IN ('schema:project_management','schema:worker_retirements')`).Scan(&domains); err != nil || domains != 0 {
		panic("parent contains administration domains")
	}
	if _, err := backup.Create(ctx, store, os.Args[1], os.Args[2]); err != nil {
		panic(err)
	}
	fmt.Println("Generated genuine predecessor bundle")
}
